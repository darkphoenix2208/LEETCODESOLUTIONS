# Problem  : Combine Two Tables
# Difficulty: Easy
# Tags     : Database
# URL      : https://leetcode.com/problems/combine-two-tables/
# Solved on: 2026-04-09 00:12
# ──────────────────────────────────────────────────

SELECT p.firstName,
       p.lastName,
       a.city,
       a.state
FROM Person p
LEFT JOIN Address a
  ON p.personId = a.personId;
